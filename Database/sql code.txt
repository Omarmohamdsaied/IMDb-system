CREATE TABLE users (
    id NUMBER PRIMARY KEY,
    name VARCHAR2(255) NOT NULL,
    email VARCHAR2(255) NOT NULL UNIQUE,
    password VARCHAR2(255) NOT NULL,
    role VARCHAR2(50) DEFAULT 'user' -- default role
);



CREATE TABLE movies (
    id NUMBER PRIMARY KEY,
    title VARCHAR2(255) NOT NULL,
    avgrating NUMBER(3,1) DEFAULT 0.0,
    release_date DATE,
    description CLOB,
    poster_url VARCHAR2(500)
);














CREATE TABLE actors (
    id NUMBER PRIMARY KEY,
    name VARCHAR2(255) NOT NULL
);

CREATE SEQUENCE actors_seq START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE users_seq START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE movies_seq START WITH 1 INCREMENT BY 1;



CREATE TABLE actor_movies (
    actor_id NUMBER,
    movie_id NUMBER,
    PRIMARY KEY (actor_id, movie_id),
    FOREIGN KEY (actor_id) REFERENCES actors(id) ON DELETE CASCADE,
    FOREIGN KEY (movie_id) REFERENCES movies(id) ON DELETE CASCADE
);




CREATE TABLE reviews (
    id NUMBER PRIMARY KEY,
    user_id NUMBER,
    movie_id NUMBER,
    text CLOB,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (movie_id) REFERENCES movies(id) ON DELETE CASCADE
);





CREATE TABLE watchlist (
    user_id NUMBER,
    movie_id NUMBER,
    PRIMARY KEY (user_id, movie_id),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (movie_id) REFERENCES movies(id) ON DELETE CASCADE
);

CREATE OR REPLACE PROCEDURE get_movie_details (
    p_movie_id       IN  NUMBER,
    p_title          OUT VARCHAR2,
    p_avgrating      OUT NUMBER,
    p_release_date   OUT DATE,
    p_description    OUT CLOB
)
AS
BEGIN
    SELECT title, avgrating, release_date, description
    INTO p_title, p_avgrating, p_release_date, p_description
    FROM movies
    WHERE id = p_movie_id;
END;

CREATE OR REPLACE PROCEDURE check_like_exists (
    p_movie_id   IN  NUMBER,
    p_user_id    IN  NUMBER,
    p_exists     OUT NUMBER
)
AS
BEGIN
    SELECT COUNT(*)
    INTO p_exists
    FROM likes
    WHERE movie_id = p_movie_id AND user_id = p_user_id;
END;
